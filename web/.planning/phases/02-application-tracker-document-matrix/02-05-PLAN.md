    Props: `slug: string`, `initialNotes?: string`.

    - Textarea element with `text-caption` font, `bg-secondary-background`, `border-2 border-border`, `p-3`, `w-full`, `min-h-[80px]`, `resize-y`
    - Placeholder: "Add notes about this application..." (from Copywriting Contract)

    - Controlled component: local state initialized from `initialNotes`
    - Debounced save: use `useEffect` with 500ms `setTimeout` to call `updateNotes(slug, localNotes)` from the tracker store. Clear timeout on unmount and on each keystroke.
    - Max length enforced: `maxLength={500}` on textarea + visual count
    - When count > 450: count text turns `text-destructive` as warning

    ```typescript
    const [localNotes, setLocalNotes] = useState(initialNotes ?? "");
    const updateNotes = useTrackerStore(s => s.updateNotes);

    useEffect(() => {
      const timer = setTimeout(() => {
        updateNotes(slug, localNotes);
      }, 500);
      return () => clearTimeout(timer);
    }, [localNotes, slug, updateNotes]);
    ```

    const docTypes = customRequirements ?? [...DOCUMENT_TYPES];

          const checked = documentChecks[docType] ?? false;
          const label = DOCUMENT_TYPE_LABELS[docType as DocumentType] ?? docType;
          return (
            <div key={docType} className="flex items-center gap-3 px-3 min-h-[44px]">
              <input
                type="checkbox"
                id={`doc-${slug}-${docType}`}
                checked={checked}
                disabled={readOnly}
                onChange={(e) => toggleDocument(slug, docType, e.target.checked)}
                className="size-5 accent-main"
              />
              <label htmlFor={`doc-${slug}-${docType}`} className="text-caption cursor-pointer flex-1">
                {label}
              </label>
            </div>
          );
        })}
        <p className="text-caption text-foreground/60 px-3 pt-2">
          {Object.values(documentChecks).filter(Boolean).length} of {docTypes.length} documents prepared
        </p>
