    ```typescript
    import type { MigrationConfig, StorageAdapter, StorageWriteResult, VersionedData } from "./types";
    import { isQuotaExceededError } from "./errors";
    import { migrateData } from "./migrate";

    // --- Types (extracted from routes/shortlist.tsx) ---

    export type Tier = "dream" | "target" | "safety";

    export interface ShortlistUniversity {
      id: string;
      name: string;
      tier: Tier;
      countries: string[];
      scholarshipCount: number;
      isCustom: boolean;
    }

    export interface ShortlistCountry {
      code: string;
      name: string;
    }

    export interface ShortlistData {
      universities: ShortlistUniversity[];
      countries: ShortlistCountry[];
    }

    type VersionedShortlistData = ShortlistData & VersionedData;

    // --- Constants ---

    export const SHORTLIST_KEY = "scholarhub-shortlist";
    const CURRENT_VERSION = 1;

    const shortlistMigrations: MigrationConfig = {
      currentVersion: CURRENT_VERSION,
      migrations: {
        1: (old) => ({
          universities: Array.isArray(old.universities) ? old.universities : [],
          countries: Array.isArray(old.countries) ? old.countries : [],
          _version: 1,
        }),
      },
    };

    // --- Adapter ---

    export class ShortlistStorageAdapter implements StorageAdapter<ShortlistData> {
      load(): ShortlistData | null {
        if (typeof window === "undefined") return null;
        try {
          const stored = localStorage.getItem(SHORTLIST_KEY);
          if (!stored) return null;
          const parsed = JSON.parse(stored);
          const migrated = migrateData<VersionedShortlistData>(parsed, shortlistMigrations);
          if (!migrated) return null;
          // Lazy migration: write back migrated data
          if (!parsed._version || parsed._version < CURRENT_VERSION) {
            try { localStorage.setItem(SHORTLIST_KEY, JSON.stringify(migrated)); } catch { /* best effort */ }
          }
          const { _version, ...data } = migrated;
          return data as ShortlistData;
        } catch {
          return null;
        }
      }

      save(data: ShortlistData): StorageWriteResult {
        if (typeof window === "undefined") return { success: false, error: "not_supported" };
        try {
          const versioned: VersionedShortlistData = { ...data, _version: CURRENT_VERSION };
          localStorage.setItem(SHORTLIST_KEY, JSON.stringify(versioned));
          return { success: true };
        } catch (err) {
          if (isQuotaExceededError(err)) return { success: false, error: "quota_exceeded" };
          return { success: false, error: "security_error" };
        }
      }

      clear(): void {
        if (typeof window === "undefined") return;
        try { localStorage.removeItem(SHORTLIST_KEY); } catch { /* SSR/disabled */ }
      }

      has(): boolean {
        if (typeof window === "undefined") return false;
        try { return localStorage.getItem(SHORTLIST_KEY) !== null; } catch { return false; }
      }
    }

    export const shortlistStorage: StorageAdapter<ShortlistData> = new ShortlistStorageAdapter();
    ```

    ```typescript
    import { useState, useEffect, useCallback } from "react";
    import type { StorageError } from "@/lib/storage/types";
    import { shortlistStorage, SHORTLIST_KEY, type ShortlistData } from "@/lib/storage/shortlist-storage";

    const DEFAULT_DATA: ShortlistData = { universities: [], countries: [] };

    export function useShortlist(): [
      ShortlistData,
      (value: ShortlistData | ((prev: ShortlistData) => ShortlistData)) => void,
      StorageError | null,
    ] {
      const [data, setDataState] = useState<ShortlistData>(() => {
        return shortlistStorage.load() ?? DEFAULT_DATA;
      });
      const [error, setError] = useState<StorageError | null>(null);

      const setData = useCallback(
        (value: ShortlistData | ((prev: ShortlistData) => ShortlistData)) => {
          setDataState((prev) => {
            const next = typeof value === "function" ? value(prev) : value;
            const result = shortlistStorage.save(next);
            if (result.success) {
              setError(null);
            } else {
              setError(result.error);
            }
            return next;
          });
        },
        [],
      );

      // Cross-tab sync via storage event
      useEffect(() => {
        const handler = (e: StorageEvent) => {
          if (e.key === SHORTLIST_KEY && e.storageArea === localStorage) {
            const loaded = shortlistStorage.load();
            setDataState(loaded ?? DEFAULT_DATA);
          }
          // Handle clear() — key is null
          if (e.key === null) {
            setDataState(DEFAULT_DATA);
          }
        };
        window.addEventListener("storage", handler);
        return () => window.removeEventListener("storage", handler);
      }, []);

      return [data, setData, error];
    }
    ```
